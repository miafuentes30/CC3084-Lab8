for antiguo, (nuevo, nombres) in GRUPOS_SQL.items():
    for origen in sorted((RAIZ / "sql" / antiguo).glob("*.sql")):
        n = int(origen.name[:2])
        CAMBIOS[origen.relative_to(RAIZ).as_posix()] = f"sql/{nuevo}/{n:02d}_{nombres[n - 1]}.sql"


def transformar(texto):
    # Las rutas del material original siguen siendo evidencia historica.
    referencias = []
    def proteger(match):
        referencias.append(match[0])
        return f"__REFERENCIA_CONSERVADA_{len(referencias) - 1}__"
    texto = re.sub(r"(?<!scripts/)lab8/(?:docs|scripts|notebooks|sql|data)/[^\s`\"'<>),]+", proteger, texto)
    for viejo, nuevo in sorted(CAMBIOS.items(), key=lambda par: -len(par[0])):
        texto = texto.replace(viejo, nuevo)
    for viejo, nuevo in CAMBIOS.items():
        if viejo.endswith((".sql", ".ipynb", ".md")) or viejo.startswith(("scripts/", "tests/")):
            texto = texto.replace(Path(viejo).name, Path(nuevo).name)
    for viejo, nuevo in MODULOS.items():
        texto = texto.replace(f"lab8.{viejo}", f"laboratorio.{nuevo}")
        texto = re.sub(rf"(from\s+)\.{viejo}(\s+import\b)", rf"\1.{nuevo}\2", texto)
    texto = texto.replace("scripts/lab8", "scripts/laboratorio")
    texto = texto.replace('"lab8.', '"laboratorio.').replace("'lab8.", "'laboratorio.")
    texto = re.sub(r"\bfrom fixtures import\b", "from datos_prueba import", texto)
    for viejo, (nuevo, _) in GRUPOS_SQL.items():
        texto = texto.replace(f"sql/{viejo}", f"sql/{nuevo}")
        texto = texto.replace(f'"{viejo}/', f'"{nuevo}/').replace(f"'{viejo}/", f"'{nuevo}/")
        texto = texto.replace(f"{viejo}/", f"{nuevo}/")
    for viejo, nuevo in CUADERNOS.items():
        texto = texto.replace(viejo, nuevo)
    texto = texto.replace('"ej[34]/*.sql"', '"[ie]*/*.sql"')
    for n, referencia in enumerate(referencias):
        texto = texto.replace(f"__REFERENCIA_CONSERVADA_{n}__", referencia)
    return texto


def main():
    if not (RAIZ / "scripts/lab8").is_dir():
        raise SystemExit("El renombramiento ya se ejecuto; no se vuelve a aplicar")
    contenidos = {}
    for origen in [RAIZ / "README.md", *[r for carpeta in ("scripts", "notebooks", "sql", "docs", "tests")
                    for r in (RAIZ / carpeta).rglob("*") if r.suffix in (".py", ".ipynb", ".sql", ".md")]]:
        texto = origen.read_text(encoding="utf-8")
        if origen.suffix == ".ipynb":
            nb = json.loads(texto)
            for celda in nb["cells"]:
                celda["source"] = transformar("".join(celda["source"])).splitlines(keepends=True)
            texto = json.dumps(nb, ensure_ascii=False, indent=1) + "\n"
        else:
            texto = transformar(texto)
        destino = RAIZ / CAMBIOS.get(origen.relative_to(RAIZ).as_posix(), origen.relative_to(RAIZ).as_posix())
        contenidos[destino] = texto
    for viejo, nuevo in CAMBIOS.items():
        origen, destino = RAIZ / viejo, RAIZ / nuevo
        if destino.exists():
            raise SystemExit(f"Destino ya existe: {destino}")
        destino.parent.mkdir(parents=True, exist_ok=True)
        origen.rename(destino)
    for destino, texto in contenidos.items():
        destino.write_text(texto, encoding="utf-8")
    # Solo elimina directorios vacios del codigo activo.
    for carpeta in [RAIZ / "scripts/lab8", *[RAIZ / "sql" / nombre for nombre in GRUPOS_SQL]]:
        if carpeta.is_dir() and not any(carpeta.iterdir()):
            carpeta.rmdir()
    (RAIZ / "docs/mapa_renombramientos.json").write_text(json.dumps(CAMBIOS, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(f"{len(CAMBIOS)} archivos renombrados; referencia lab8/ intacta")


if __name__ == "__main__":
    main()
